local player = game.Players.LocalPlayer
local replicatedStorage = game:GetService("ReplicatedStorage")
local RS = replicatedStorage

local NINJA_PLACE = 200
local NINJA_UPGRADES = {200, 300, 400, 600}

local HELI_PLACE = 200
local HELI_UPGRADES = {300, 500, 800, 1500}

local ninjaValue = 0
local heliValue = 0
local autoFarmStarted = false

--========================
-- MONEY
--========================

local function getMoney()

    local stats = player:FindFirstChild("leaderstats")
    local money = stats and stats:FindFirstChild("Money")

    return money and money.Value or 0
end

local function waitMoney(amount)

    while getMoney() < amount do
        task.wait(0.5)
    end
end

--========================
-- REMOTES
--========================

local function place(unitName, position, slot)

    local args = {
        [1] = {
            [1] = {
                [1] = "\226\129\130\22",
                [2] = unitName,
                [3] = position,
                [4] = slot
            }
        }
    }

    pcall(function()
        RS.NetworkingContainer.DataRemote:FireServer(unpack(args))
    end)
end

local function upgrade(unit)

    if not unit then
        return
    end

    local args = {
        [1] = {
            [1] = {
                [1] = "\226\129\130#",
                [2] = unit
            }
        }
    }

    pcall(function()
        RS.NetworkingContainer.DataRemote:FireServer(unpack(args))
    end)
end

--========================
-- UNIT FINDER
--========================

local function getUnitByValue(unitName, num)

    local troops = workspace:FindFirstChild("Troops")

    if not troops then
        return nil
    end

    for _, v in ipairs(troops:GetChildren()) do

        if v.Name == unitName then

            local val = v:FindFirstChild("Value")

            if not val then

                val = Instance.new("IntValue")
                val.Name = "Value"
                val.Parent = v

                local count = 0

                for _, x in ipairs(troops:GetChildren()) do
                    if x.Name == unitName then
                        count += 1
                    end
                end

                val.Value = count
            end

            if val.Value == num then
                return v
            end
        end
    end

    return nil
end

--========================
-- NINJA
--========================

local function placeAndMaxNinja(position)

    waitMoney(NINJA_PLACE)

    ninjaValue += 1

    repeat
        place("NinjaCameraman", position, 1)
        task.wait(1)
    until getUnitByValue("NinjaCameraman", ninjaValue)

    for _, cost in ipairs(NINJA_UPGRADES) do

        waitMoney(cost)

        local target = getUnitByValue("NinjaCameraman", ninjaValue)

        if target then
            upgrade(target)
        end

        task.wait(0.5)
    end
end

--========================
-- HELI
--========================

local function placeAndMaxHeli(position)

    waitMoney(HELI_PLACE)

    heliValue += 1

    repeat
        place("CameraHelicopter", position, 5)
        task.wait(1)
    until getUnitByValue("CameraHelicopter", heliValue)

    for _, cost in ipairs(HELI_UPGRADES) do

        waitMoney(cost)

        local target = getUnitByValue("CameraHelicopter", heliValue)

        if target then
            upgrade(target)
        end

        task.wait(0.5)
    end
end

--========================

local function startInGameScript()

    if autoFarmStarted then
        return
    end

    autoFarmStarted = true

    task.wait(2)

    ninjaValue = 0
    heliValue = 0

    local lane1 = Vector3.new(-69.5605, 2.2020, 55.6358)
    local lane2 = lane1 + Vector3.new(5,0,0)

    -- Ninja 13 à¸•à¸±à¸§

        placeAndMaxNinja(lane1)
        placeAndMaxNinja(lane2)

        lane1 += Vector3.new(0,0,4)
        lane2 += Vector3.new(0,0,4)
    end

    placeAndMaxNinja(lane1)
